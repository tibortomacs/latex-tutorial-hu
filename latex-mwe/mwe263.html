% !TEX lualatex

\documentclass{article}
\usepackage{fontspec,luacode}

\begin{luacode}
function rows(n)
  for i = 0, n do
    x = 0.1 * i
    y = math.sin(x)
    z = math.cos(x)
    y4 = string.format("%.4f", y)
    z4 = string.format("%.4f", z)
    tex.sprint(x .. " & " .. y4 .. " & " .. z4 .. " \\\\")
  end
end
\end{luacode}

\def\sincostable#1{%
  \begin{tabular}{@{}ccc@{}}
    $x$ & $\sin(x)$ & $\cos(x)$ \\ \hline
    \directlua{rows(#1)}
  \end{tabular}}

\begin{document}

\sincostable{10}

\end{document}
